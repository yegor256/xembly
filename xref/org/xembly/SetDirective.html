5   package org.xembly;
6   
7   import lombok.EqualsAndHashCode;
8   import org.w3c.dom.Node;
9   
10  /**
11   * SET directive.
12   *
13   * <p>The class is immutable and thread-safe.</p>
14   *
15   * @since 0.1
16   */
17  @EqualsAndHashCode(of = "value")
18  final class SetDirective implements Directive {
19  
20      /**
21       * Text value to set.
22       */
23      private final Arg value;
24  
25      /**
26       * Public ctor.
27       *
28       * @param val Text value to set
29       * @throws XmlContentException If invalid input
30       */
31      SetDirective(final String val) throws XmlContentException {
32          this.value = new Arg(val);
33      }
34  
35      @Override
36      public String toString() {
37          return String.format("SET %s", this.value);
38      }
39  
40      @Override
41      public Directive.Cursor exec(final Node dom,
42          final Directive.Cursor cursor, final Directive.Stack stack) {
43          for (final Node node : cursor) {
44              node.setTextContent(this.value.raw());
45          }
46          return cursor;
47      }
48  }
