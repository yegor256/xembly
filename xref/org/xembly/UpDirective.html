5   package org.xembly;
6   
7   import java.util.Collection;
8   import java.util.HashSet;
9   import lombok.EqualsAndHashCode;
10  import org.w3c.dom.Node;
11  
12  /**
13   * UP directive.
14   *
15   * <p>The class is immutable and thread-safe.</p>
16   *
17   * @since 0.1
18   */
19  @EqualsAndHashCode
20  final class UpDirective implements Directive {
21  
22      @Override
23      public String toString() {
24          return "UP";
25      }
26  
27      @Override
28      @SuppressWarnings("aibolit.P13")
29      public Directive.Cursor exec(final Node dom,
30          final Directive.Cursor cursor, final Directive.Stack stack)
31          throws ImpossibleModificationException {
32          final Collection<Node> parents = new HashSet<>(cursor.size());
33          for (final Node node : cursor) {
34              final Node parent = node.getParentNode();
35              if (parent == null) {
36                  throw new ImpossibleModificationException(
37                      String.format(
38                          "There is no parent node of '%s' (%s), can't go UP",
39                          node.getNodeName(), node.getNodeType()
40                      )
41                  );
42              }
43              parents.add(parent);
44          }
45          return new DomCursor(parents);
46      }
47  }
