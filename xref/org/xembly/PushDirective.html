5   package org.xembly;
6   
7   import lombok.EqualsAndHashCode;
8   import org.w3c.dom.Node;
9   
10  /**
11   * PUSH directive.
12   *
13   * <p>The class is immutable and thread-safe.</p>
14   *
15   * @since 0.16
16   */
17  @EqualsAndHashCode
18  final class PushDirective implements Directive {
19  
20      @Override
21      public String toString() {
22          return "PUSH";
23      }
24  
25      @Override
26      public Directive.Cursor exec(final Node dom,
27          final Directive.Cursor cursor, final Directive.Stack stack)
28          throws ImpossibleModificationException {
29          stack.push(cursor);
30          return cursor;
31      }
32  }
