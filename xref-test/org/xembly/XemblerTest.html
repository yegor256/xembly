198     @Test
199     void escapesEverything() throws Exception {
200         final StringBuilder pool = new StringBuilder(0x20);
201         for (int chr = 0x00; chr <= 0x1F; ++chr) {
202             pool.append((char) chr);
203         }
204         for (int chr = 0x7F; chr <= 0x9F; ++chr) {
205             pool.append((char) chr);
206         }
207         pool.append("привет мир, XML; 42 <>&\"'");
208         final Random random = new Random(0x305L);
209         final DocumentBuilder builder = DocumentBuilderFactory.newInstance()
210             .newDocumentBuilder();
211         for (int iter = 0; iter < 500; ++iter) {
212             final StringBuilder value = new StringBuilder(0);
213             final int length = random.nextInt(64);
214             for (int pos = 0; pos < length; ++pos) {
215                 value.append(pool.charAt(random.nextInt(pool.length())));
216             }
217             final String raw = value.toString();
218             final String escaped = Xembler.escape(raw);
219             final String xml = new Xembler(
220                 new Directives().add("r").attr("a", escaped)
221             ).xmlQuietly();
222             try (StringReader reader = new StringReader(xml)) {
223                 MatcherAssert.assertThat(
224                     String.format("Can't escape text: %s", escaped),
225                     builder.parse(new InputSource(reader))
226                         .getDocumentElement().hasAttribute("a"),
227                     Matchers.is(true)
228                 );
229             }
230         }
231     }

285     @Test
286     void concurrentInvocationWithNoExceptions() throws Exception {
287         final Document dom = DocumentBuilderFactory.newInstance()
288             .newDocumentBuilder().newDocument();
289         dom.appendChild(dom.createElement("root"));
290         final Xembler xembler = new Xembler(
291             new Directives(
292                 "XPATH '/root';ADDIF 'blow';"
293             )
294         );
295         final int tasks = 10;
296         new LengthOf(
297             new Threads<>(
298                 tasks,
299                 new Repeated<>(
300                     () -> {
301                         xembler.apply(dom);
302                         return null;
303                     },
304                     tasks
305                 )
306             )
307         ).value();
308         MatcherAssert.assertThat(
309             "Can't run Xembler concurrently",
310             XhtmlMatchers.xhtml(dom),
311             XhtmlMatchers.hasXPaths(
312                 "/root[count(blow) = 1]"
313             )
314         );
315     }

343     @ParameterizedTest
344     @ClasspathSource(value = "org/xembly/stories/", glob = "**.yml")
345     void checksYamlStories(final String story) {
346         MatcherAssert.assertThat(
347             "modifies XML document with a few directives",
348             XemblerTest.outcomeOf(story),
349             XemblerTest.matchersOf(story)
350         );
351     }
352 
353     @SuppressWarnings("unchecked")
354     private static XML outcomeOf(final String story) {
355         final Map<String, Object> yaml = new Yaml().load(story);
356         return new XMLDocument(
357             new Xembler(
358                 new Directives(
359                     String.join(
360                         "",
361                         (Iterable<String>) yaml.get("directives")
362                     )
363                 )
364             ).applyQuietly(
365                 new XMLDocument(yaml.get("before").toString()).inner()
366             )
367         );
368     }
369 
370     @SuppressWarnings("unchecked")
371     private static Matcher<XML> matchersOf(final String story) {
372         return new AllOf<>(
373             new ListOf<>(
374                 new Mapped<>(
375                     str -> new XPathMatcher<>(str, new XPathContext()),
376                     (Collection<String>) new Yaml()
377                         .<Map<String, Object>>load(story)
378                         .get("xpaths")
379                 )
380             )
381         );
382     }
